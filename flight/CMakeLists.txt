cmake_minimum_required(VERSION 3.16)
project(oslurm_flight LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(OSLURM_ENABLE_CG "Include the float64 compute kernel" ON)
set(FLIGHT_SOURCES src/decision.cpp)
if(OSLURM_ENABLE_CG)
  list(APPEND FLIGHT_SOURCES src/cg.cpp)
endif()
add_library(oslurm_flight STATIC ${FLIGHT_SOURCES})
target_include_directories(oslurm_flight PUBLIC include)
target_compile_options(oslurm_flight PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti -ffreestanding -fstack-usage)
add_library(flight_c_example OBJECT examples/agent.c)
target_include_directories(flight_c_example PRIVATE include)
option(OSLURM_HOST_TESTS "Build host ABI and tests" ON)
if(OSLURM_HOST_TESTS)
  add_library(oslurm_flight_host SHARED src/decision.cpp src/cg.cpp)
  target_include_directories(oslurm_flight_host PUBLIC include)
  target_compile_options(oslurm_flight_host PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
  enable_testing()
  add_executable(flight_tests tests/test_flight.cpp)
  target_compile_options(flight_tests PRIVATE -UNDEBUG)
  target_link_libraries(flight_tests PRIVATE oslurm_flight)
  add_test(NAME flight COMMAND flight_tests)
  add_executable(flight_c_abi tests/c_api.c)
  target_link_libraries(flight_c_abi PRIVATE oslurm_flight)
  add_test(NAME c_abi COMMAND flight_c_abi)
  add_executable(flight_benchmark tests/benchmark.cpp)
  target_link_libraries(flight_benchmark PRIVATE oslurm_flight)
endif()
